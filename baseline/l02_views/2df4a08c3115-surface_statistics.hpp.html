   8 struct SurfaceStatistics {double ground=0,top=0;int supported_modes=0;};
   9 inline SurfaceStatistics surfaceStatistics(const std::vector<Eigen::Vector3d>& rays,
  10  double voxel_ground,double voxel_top,double floor,double sigma=.012,double strong=.045) {
  11  if(rays.empty())return {voxel_ground,voxel_top,0};
  12  std::vector<double> z;for(const auto&p:rays)z.push_back(p.z());std::sort(z.begin(),z.end());
  13  SurfaceStatistics r{z[size_t(.25*(z.size()-1))]-floor,z[size_t(.8*(z.size()-1))]-floor,0};
  14  // Strong geometric hazards keep the existing immediate upper-channel policy.
  15  if(voxel_top>=strong||voxel_top<=-strong)r.top=voxel_top;
  16  // A small, coherent raised surface must not disappear into many floor rays.
  17  // Require spatial support and a confidence-separated mode; random isolated
  18  // range-noise bins must not receive the same weight as hundreds of floor rays.
  19  std::map<std::pair<int,int>,std::vector<double>> high,low;
  20  for(const auto&p:rays){double h=p.z()-floor;auto key=std::make_pair(int(std::floor(p.x()/.04)),int(std::floor(p.y()/.04)));if(h>r.ground+2*sigma)high[key].push_back(h);if(h<r.top-2*sigma)low[key].push_back(h);}
  21  for(const auto&[key,a]:high)if(a.size()>=3){double mean=0;for(double v:a)mean+=v;mean/=a.size();if(mean-2*sigma/std::sqrt(a.size())>r.ground+2*sigma){r.top=std::max(r.top,mean);++r.supported_modes;}}
  22  for(const auto&[key,a]:low)if(a.size()>=3){double mean=0;for(double v:a)mean+=v;mean/=a.size();if(mean+2*sigma/std::sqrt(a.size())<r.top-2*sigma){r.ground=std::min(r.ground,mean);++r.supported_modes;}}
  23  return r;
  24 }
