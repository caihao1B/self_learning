   7 // Filter only the map->odom correction. Local odometry retains its full rate.
   8 // Caller must check RAW innovation, jumps, freshness and plan holds first.
   9 inline Eigen::Isometry3d smoothCorrection(const Eigen::Isometry3d &previous,
  10                                          const Eigen::Isometry3d &observed,
  11                                          double dt, double tau) {
  12   if (!previous.matrix().allFinite() || !observed.matrix().allFinite() ||
  13       !std::isfinite(dt) || !std::isfinite(tau) || tau <= 0.)
  14     throw std::invalid_argument("invalid correction filter input");
  15   if (dt <= 0.) return previous;
  16   const double alpha = -std::expm1(-dt / tau);
  17   Eigen::Isometry3d result = Eigen::Isometry3d::Identity();
  18   result.translation() = previous.translation() + alpha * (observed.translation() - previous.translation());
  19   result.linear() = Eigen::Quaterniond(previous.linear()).slerp(alpha, Eigen::Quaterniond(observed.linear())).normalized().toRotationMatrix();
  20   return result;
  21 }
