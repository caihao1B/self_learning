  23 class WheelDrive final : public gz::sim::System, public gz::sim::ISystemConfigure,
  24  public gz::sim::ISystemPreUpdate,public gz::sim::ISystemPostUpdate,public gz::sim::ISystemReset {
  25  using Wall=std::chrono::steady_clock;
  26  Json c,pending; std::mutex mutex;gz::transport::Node node;
  27  gz::transport::Node::Publisher feedback,truth;
  28  std::vector<gz::sim::Entity> joints;gz::sim::Entity body=0;
  29  uint64_t received=0,seen=0,lastSequence=0,epoch=0,resets=0;
  30  Wall::time_point receipt=Wall::now(),lastPublish=Wall::now();
  31  double lastRequestSim=-1e6,lastSim=-1,lastTruth=-1,speed=0,yaw=0,heldK=0,trackingSince=-1;
  32  bool latched=true,wasPaused=true;std::string reason="STARTUP";
  33  std::vector<double> command;std::string session,revision;bool ack=false;
  34  void latch(const std::string &r){if(!latched || reason=="STARTUP")reason=r;latched=true;}
  35  void Request(const gz::msgs::StringMsg &m){
  36    std::lock_guard<std::mutex> lock(mutex);
  37    try{pending=Json::parse(m.data());}catch(...){pending=Json();}
  38    receipt=Wall::now();++received;
  39  }
  40  bool read(const gz::sim::EntityComponentManager &ecm,std::vector<double>&q,std::vector<double>&v) const {
  41    for(auto j:joints){
  42      auto p=ecm.Component<gz::sim::components::JointPosition>(j);auto w=ecm.Component<gz::sim::components::JointVelocity>(j);
  43      if(!p||!w||p->Data().empty()||w->Data().empty()||!std::isfinite(p->Data()[0])||!std::isfinite(w->Data()[0]))return false;
  44      q.push_back(p->Data()[0]);v.push_back(w->Data()[0]);
  45    }return true;
  46  }
  47  public:
  48  void Configure(const gz::sim::Entity &entity,const std::shared_ptr<const sdf::Element> &sdf,gz::sim::EntityComponentManager &ecm,gz::sim::EventManager &) override {
  49    c=Json::parse(sdf->Get<std::string>("config"));session=sdf->Get<std::string>("session");revision=c.at("config_hash");ack=c.at("kind")=="ackermann";
  50    gz::sim::Model m(entity);body=m.LinkByName(ecm,"base_link");
  51    std::vector<std::string> names={"rear_left_joint","rear_right_joint"};if(ack){names.push_back("steer_left_joint");names.push_back("steer_right_joint");}else{names.push_back("front_left_joint");names.push_back("front_right_joint");}
  52    for(auto &name:names){auto j=m.JointByName(ecm,name);if(!j)throw std::runtime_error("missing physical joint "+name);
  53      joints.push_back(j);ecm.CreateComponent(j,gz::sim::components::JointPosition());ecm.CreateComponent(j,gz::sim::components::JointVelocity());}
  54    command.resize(joints.size(),0);
  55    node.Subscribe("/s03/actuator/request",&WheelDrive::Request,this);
  56    feedback=node.Advertise<gz::msgs::StringMsg>("/s03/encoders");truth=node.Advertise<gz::msgs::StringMsg>("/s03/evaluation/pose");
  57  }
  58  void Reset(const gz::sim::UpdateInfo &,gz::sim::EntityComponentManager &ecm) override {
  59    ++epoch;++resets;lastSequence=0;lastRequestSim=-1e6;lastSim=-1;lastTruth=-1;latched=true;reason="RESET_REQUIRES_ARM";speed=yaw=heldK=0;trackingSince=-1;
  60    for(auto j:joints)ecm.SetComponentData<gz::sim::components::JointVelocityCmd>(j,{0});
  61  }
  62  void PreUpdate(const gz::sim::UpdateInfo &info,gz::sim::EntityComponentManager &ecm) override {
  63    const double t=std::chrono::duration<double>(info.simTime).count();
  64    const double dt=std::chrono::duration<double>(info.dt).count();
  65    if(info.paused&&!wasPaused){++epoch;lastSequence=0;latched=true;reason="PAUSED_REQUIRES_ARM";speed=yaw=0;}
  66    wasPaused=info.paused;
  67    if(lastSim>=0 && t<lastSim){++epoch;latched=true;reason="CLOCK_REWIND_REQUIRES_ARM";speed=yaw=0;}
  68    lastSim=t;
  69    std::vector<double> q,vel;bool valid=read(ecm,q,vel);
  70    const double r=c["wheel"]["radius"],T=c.contains("calibration")?c["calibration"].value("effective_track_m",c["wheel"]["track"].get<double>()):c["wheel"]["track"].get<double>(),L=c["wheelbase"];
  71    const double driveTrack=c.contains("calibration")?c["calibration"].value("drive_track_m",T):T;
  72    bool stopped=valid && std::abs(vel[0]*r)<.01 && std::abs(vel[1]*r)<.01;
  73    Json request;uint64_t seq;double age;
  74    {std::lock_guard<std::mutex> lock(mutex);request=pending;seq=received;age=std::chrono::duration<double>(Wall::now()-receipt).count();}
  75    if(seq!=seen){seen=seq;
  76      try {
  77        if(request.at("session")!=session || request.at("revision")!=revision || request.at("epoch").get<uint64_t>()!=epoch || request.at("sequence").get<uint64_t>()<=lastSequence)throw std::runtime_error("LEASE_ID_OR_REPLAY");
  78        double stamp=request.at("stamp"),v=request.at("v"),w=request.at("w");
  79        if(!std::isfinite(stamp)||!std::isfinite(v)||!std::isfinite(w)||t-stamp<-.03||t-stamp>.2)throw std::runtime_error("INVALID_OR_STALE_STAMP");
  80        lastSequence=request.at("sequence");lastRequestSim=t;
  81        std::string mode=request.at("mode");
  82        if(mode=="STOP")latch("STOP_REQUEST");
  83        else if(mode=="ARM" && !info.paused && stopped && std::abs(v)<.001 && std::abs(w)<.001){latched=false;reason="ARMED";}
  84        else if(mode!="DRIVE" && mode!="ARM")latch("INVALID_MODE");
  85      }catch(const std::exception &){latch("LEASE_REJECTED");}
  86    }
  87    if(age>c["safety"]["actuator_wall_s"].get<double>() || t-lastRequestSim>c["safety"]["actuator_sim_s"].get<double>())latch("ACTUATOR_WATCHDOG");
  88    if(!valid)latch("ENCODER_MISSING");
  89    double v=0,w=0;
  90    if(!latched && !info.paused && request.is_object() && request.value("mode",std::string())=="DRIVE"){
  91      v=request.value("v",0.);w=request.value("w",0.);
  92      v=std::clamp(v,-c["limits"]["reverse"].get<double>(),c["limits"]["max_speed"].get<double>());
  93      w=std::clamp(w,-c["limits"]["yaw_rate"].get<double>(),c["limits"]["yaw_rate"].get<double>());
  94    }
  95    if(ack && std::abs(v)<.001){v=w=0;}
  96    if(!info.paused && dt>0 && dt<.1 && valid){
  97      double k=heldK,error=0;
  98      if(ack && v!=0)k=std::clamp(w/v,-c["steering"]["max_curvature"].get<double>(),c["steering"]["max_curvature"].get<double>());
  99      if(ack){heldK=k;double al=atan(L*k/(1-T*k/2)),ar=atan(L*k/(1+T*k/2));error=std::max(abs(al-q[2]),abs(ar-q[3]));
 100        if(error>c["safety"]["alignment_error_rad"].get<double>()){
 101          v=std::clamp(v,-c["safety"]["alignment_speed_mps"].get<double>(),c["safety"]["alignment_speed_mps"].get<double>());
 102          if(trackingSince<0)trackingSince=t;
 103          if(t-trackingSince>c["safety"]["steering_tracking_timeout_s"].get<double>()){latch("STEERING_TRACKING_FAULT");v=0;}
 104        }else trackingSince=-1;
 105        const double gain=c["steering"]["gain"],rate=c["steering"]["rate_rad_s"];
 106        command[2]=std::clamp(gain*(al-q[2]),-rate,rate);command[3]=std::clamp(gain*(ar-q[3]),-rate,rate);
 107      }
 108      if(speed*v<0)v=0;
 109      const double rate=(abs(v)<abs(speed))?c["limits"]["deceleration"].get<double>():c["limits"]["acceleration"].get<double>();
 110      speed+=std::clamp(v-speed,-rate*dt,rate*dt);
 111      const double wrate=c["limits"]["yaw_acceleration"];yaw+=std::clamp(w-yaw,-wrate*dt,wrate*dt);
 112      if(ack){double a=tan(q[2]),b=tan(q[3]);k=.5*(a/(L+T*a/2)+b/(L-T*b/2));yaw=speed*k;}
 113      command[0]=(speed-driveTrack*yaw/2)/r;command[1]=(speed+driveTrack*yaw/2)/r;
 114      if(!ack){command[2]=command[0];command[3]=command[1];}
 115    }else if(info.paused || !valid){std::fill(command.begin(),command.end(),0);}
 116    for(size_t i=0;i<joints.size();++i)ecm.SetComponentData<gz::sim::components::JointVelocityCmd>(joints[i],{command[i]});
 117    if(std::chrono::duration<double>(Wall::now()-lastPublish).count()>=.02){
 118      lastPublish=Wall::now();Json d={{"t",t},{"epoch",epoch},{"resets",resets},{"q",q},{"velocity",vel},{"command",command},{"armed",!latched},{"reason",reason},{"paused",info.paused},{"limited_v",speed},{"limited_w",yaw},{"valid",valid}};
 119      gz::msgs::StringMsg m;m.set_data(d.dump());feedback.Publish(m);
 120    }
 121  }
 122  void PostUpdate(const gz::sim::UpdateInfo &info,const gz::sim::EntityComponentManager &ecm) override {
 123    const double t=std::chrono::duration<double>(info.simTime).count();if(info.paused||t-lastTruth<.019)return;lastTruth=t;
 124    const auto p=gz::sim::worldPose(body,ecm);Json d={{"t",t},{"epoch",epoch},{"resets",resets},{"xyz",{p.X(),p.Y(),p.Z()}},{"rpy",{p.Rot().Roll(),p.Rot().Pitch(),p.Rot().Yaw()}}};
 125    gz::msgs::StringMsg m;m.set_data(d.dump());truth.Publish(m);
 126  }
