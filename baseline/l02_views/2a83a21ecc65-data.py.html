   8 def json_value(value):
   9     """ROS/numpy scalar values and invalid measurements become valid JSON."""
  10     if isinstance(value,dict): return {k:json_value(v) for k,v in value.items()}
  11     if isinstance(value,(list,tuple)): return [json_value(v) for v in value]
  12     if hasattr(value,'item'): return json_value(value.item())
  13     if isinstance(value,float) and not math.isfinite(value): return None
  14     return value

  16 def read_json(path, limit=8_000_000):
  17     try:
  18         if path.stat().st_size > limit:
  19             return {}
  20         value = json.loads(path.read_text())
  21         return value if isinstance(value, dict) else {}
  22     except (OSError, ValueError):
  23         return {}

  25 def tail_records(path, limit=131072, count=120):
  26     """Ignore an unfinished last line; never read an unbounded runtime log."""
  27     try:
  28         with path.open('rb') as stream:
  29             size = stream.seek(0, 2)
  30             start = max(0, size-limit)
  31             stream.seek(start)
  32             raw = stream.read(limit)
  33         lines = raw.split(b'\n')
  34         if start:
  35             lines = lines[1:]
  36         lines = lines[:-1]
  37         rows = []
  38         for line in lines:
  39             try:
  40                 row = json.loads(line)
  41                 if isinstance(row, dict): rows.append(row)
  42             except (ValueError, UnicodeError):
  43                 pass
  44         return rows[-count:], start == 0 and len(rows) <= count
  45     except OSError:
  46         return [], False

  48 def last(path, key=None, value=None):
  49     rows, _ = tail_records(path)
  50     return next((r for r in reversed(rows) if key is None or r.get(key) == value), {})

  52 def age(path):

  56 def freshness(stamp, now, maximum=.5):
  57     if not isinstance(stamp, (int,float)) or not isinstance(now, (int,float)):
  58         return 'MISSING', None
  59     delta = now-stamp
  60     if not math.isfinite(delta): return 'MISSING', None
  61     return ('FUTURE' if delta < -.1 else 'FRESH' if delta <= maximum else 'STALE'), delta

  68     def runs(self):
  69         if time.monotonic()-self._listed < 3: return self._runs
  70         runs=[]
  71         for stage in STAGES:
  72             parent=self.results/stage
  73             if not parent.is_dir(): continue
  74             for p in parent.iterdir():
  75                 if not p.is_dir() or not (p/'spec.json').is_file(): continue
  76                 spec=read_json(p/'spec.json')
  77                 if spec.get('task') not in ('spray','inspection','mowing'): continue
  78                 stamp=p.name.split('_',1)[-1]
  79                 runs.append(dict(id=stage+'/'+p.name, task=spec['task'], case=spec.get('case',''), sort=stamp))
  80         self._runs=sorted(runs,key=lambda r:r['sort'],reverse=True)[:160]
  81         self._listed=time.monotonic(); return self._runs

  83     def resolve(self, name='auto'):
  84         runs=self.runs()
  85         if name=='auto':
  86             # Prefer a live writer over a newly created but not yet started directory.
  87             for r in runs:
  88                 p=self.results/r['id']
  89                 if not (p/'runtime.json').exists() and (age(p/'live.json') or 1e9)<8:
  90                     return p,r['id']
  91             if not runs: return None,None
  92             name=runs[0]['id']
  93         if name not in {r['id'] for r in runs}: raise ValueError('Unknown run')
  94         p=(self.results/name).resolve()
  95         if not p.is_relative_to(self.results.resolve()): raise ValueError('Run outside results')
  96         return p,name
