   4 def terminal_straight(goal,distance=1.2,spacing=.04):
   5     """Same-pose arrival with a straight tangent before it; proposal, not admission."""
   6     if len(goal)!=3 or not all(math.isfinite(v) for v in goal) or not 0<distance<=1.2 or not 0<spacing<=.1:
   7         raise ValueError('INVALID_TERMINAL_APPROACH')
   8     x,y,h=goal;count=math.ceil(distance/spacing)
   9     return [[x-distance*(1-i/count)*math.cos(h),y-distance*(1-i/count)*math.sin(h),h] for i in range(count+1)]

  11 def retreat_prefix(start,distance=.6,spacing=.04):
  12     """Proposal only: a straight tool-off retreat, later swept/costmap checked."""
  13     if not all(math.isfinite(v) for v in start) or not 0<distance<=.8 or spacing<=0:
  14         raise ValueError('INVALID_RETREAT_BOUND')

  16     return [[x-distance*i/count*math.cos(h),y-distance*i/count*math.sin(h),h] for i in range(count+1)]

  18 def gear_runs(points):
  19     """Split at a sign change of motion along body heading; share the cusp pose."""
  20     runs=[];current=[];previous=0
  21     for a,b in zip(points,points[1:]):
  22         dx,dy=b[0]-a[0],b[1]-a[1]
  23         if math.hypot(dx,dy)<1e-7:continue
  24         heading=a[2]+.5*math.atan2(math.sin(b[2]-a[2]),math.cos(b[2]-a[2]))
  25         longitudinal=dx*math.cos(heading)+dy*math.sin(heading)
  26         if abs(longitudinal)<.9*math.hypot(dx,dy):raise ValueError('LATERAL_ACKERMANN_PATH')
  27         direction=1 if longitudinal>0 else -1
  28         if direction!=previous:
  29             if current:runs.append(dict(direction=previous,poses=current))
  30             current=[a];previous=direction
  31         current.append(b)
  32     if current:runs.append(dict(direction=previous,poses=current))
  33     if not runs:raise ValueError('NO_TRANSLATION')
  34     return runs

  36 def tracking_distance(points,pose,cursor):
  37     """Distance to local traversed/current edges, not an overlapping future branch."""
  38     edges=list(zip(points[max(0,cursor-2):cursor+4],points[max(0,cursor-2)+1:cursor+5]))
  39     distances=[]
  40     for a,b in edges:
  41         dx,dy=b[0]-a[0],b[1]-a[1];den=dx*dx+dy*dy
  42         u=max(0.,min(1.,((pose[0]-a[0])*dx+(pose[1]-a[1])*dy)/den)) if den else 0.
  43         distances.append(math.hypot(pose[0]-a[0]-u*dx,pose[1]-a[1]-u*dy))
  44     return min(distances,default=math.dist(points[cursor][:2],pose[:2]))

  56 def observed_cursor(points,pose,cursor,window_m=.20):
  57     """Advance only through a local prefix, never jump to a nearby return branch."""
  58     end=cursor;distance=0.

  64 def validate_candidate(points,start,goal,cfg,max_length=12.):
  65     from shapely.geometry import shape
  66     from footprint_r09 import body_polygon
  67     if len(points)<2 or any(not all(math.isfinite(v) for v in p) for p in points):return False,'MALFORMED_PATH'
  68     if math.dist(points[0][:2],start[:2])>.15 or yaw_error(points[0][2],start[2])>.10:return False,'START_DISCONTINUITY'
  69     if math.dist(points[-1][:2],goal[:2])>.05 or yaw_error(points[-1][2],goal[2])>.08:return False,'GOAL_MISMATCH'
  70     # The authoritative prior is in world coordinates; navigation is ENU.
  71     allowed=shape(cfg['allowed_prior']).buffer(-cfg['footprint_margin_m'])
  72     for p in points:
  73         if not allowed.covers(body_polygon(cfg['vehicle']['footprint'],p[0]+28,p[1]+14,p[2])):return False,'FOOTPRINT_OUTSIDE_PRIOR'
  74     length=0.
  75     for a,b in zip(points,points[1:]):
  76         ds=math.dist(a[:2],b[:2]);angle=yaw_error(a[2],b[2]);length+=ds
  77         if ds>.4:return False,'PATH_DISCONTINUITY'
  78         # Arc-length curvature from chord; identical cusp points may keep yaw.
  79         if ds<1e-5:
  80             if angle>1e-4:return False,'IN_PLACE_ROTATION'
  81         elif 2*math.sin(angle/2)/ds>cfg['vehicle']['steering']['max_curvature']+.005:return False,'CURVATURE_EXCEEDED'

  83         for j in range(1,steps):

  85             if not allowed.covers(body_polygon(cfg['vehicle']['footprint'],p[0]+28,p[1]+14,p[2])):return False,'SWEPT_FOOTPRINT_OUTSIDE_PRIOR'
  86     if not math.isfinite(max_length) or max_length<=0:return False,'INVALID_LENGTH_BOUND'
  87     if length>max_length:return False,'RECOVERY_TOO_LONG'
  88     return True,'PRIOR_AND_KINEMATICS_VALID_OBSERVED_RISK_STILL_REQUIRED'
