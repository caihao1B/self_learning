  12 def validate(plan, original):
  13     old=original['clusters'][0]; new=plan['cluster']; segments=new['segments']
  14     old_work=[s for s in old['segments'] if s['kind']=='COVERAGE']
  15     original_by_id={s['id']:s for s in old_work}
  16     old_union=union_all([Polygon(s['target_area']) for s in old_work])
  17     retained=union_all([Polygon(s['target_area']) for s in new['original_work_segments']])
  18     if new['original_work_segments']!=old_work or old_union.symmetric_difference(retained).area>1e-8:
  19         raise ValueError('ORIGINAL_WORK_POLYGONS_CHANGED')
  20     if new['work_area_m2']!=old['work_area_m2'] or new['area_map']!=old['area_map']:
  21         raise ValueError('ORIGINAL_CLUSTER_AREA_CHANGED')
  22     if new['budget_s']!=old['budget_s'] or new['estimated_seconds']>old['budget_s']:
  23         raise ValueError('ORIGINAL_TIME_BUDGET_CHANGED_OR_EXCEEDED')
  24     prior=shape(plan['navigation_prior']); allowed=prior.buffer(-original['footprint_margin_m'])
  25     if prior.difference(shape(original['allowed_prior'])).area>1e-8:
  26         raise ValueError('NAVIGATION_AUTHORITY_EXPANDED')
  27     nav=yaml.safe_load((edition('R09')/'nav2.yaml').read_text())
  28     maps=[nav[k][k]['ros__parameters'] for k in ['local_costmap','global_costmap']]
  29     expected,policy=make_envelope(original['vehicle']['footprint'],max(c['footprint_padding'] for c in maps),max(c['resolution'] for c in maps))
  30     if plan.get('planning_envelope')!=expected or plan.get('planning_clearance')!=policy:
  31         raise ValueError('GRID_AND_TRACKING_ENVELOPE_CHANGED')
  32     cfg=copy.deepcopy(original);cfg['allowed_prior']=plan['navigation_prior']
  33     previous=None
  34     for s in segments:
  35         if previous and (math.dist(previous[:2],s['poses'][0][:2])>1e-6 or yaw_error(previous[2],s['poses'][0][2])>1e-6):
  36             raise ValueError('DISCONNECTED_EXECUTION_PATH')
  37         valid,why=validate_candidate(s['poses'],s['poses'][0],s['poses'][-1],cfg,60.)
  38         if not valid:raise ValueError(why)
  39         runs=gear_runs(s['poses'])
  40         if len(runs)!=1 or runs[0]['direction']!=s['direction']:raise ValueError('UNDECLARED_GEAR_CHANGE')
  41         for p,w in zip(s['poses'],s['poses_world']):
  42             if math.dist([p[0]+28,p[1]+14,p[2]],w)>1e-7:raise ValueError('FRAME_MISMATCH')
  43             if not allowed.covers(body_polygon(plan['planning_envelope'],*w)):raise ValueError('PLANNING_ENVELOPE_OUTSIDE_PRIOR')
  44         if s['kind']=='COVERAGE':
  45             source=original_by_id[s['work_key']]
  46             if s['target_area']!=source['target_area'] or s['requested_area_m2']!=source['requested_area_m2'] or s['work_target_pose']!=source['poses'][0]:
  47                 raise ValueError('FEEDBACK_OBLIGATION_CHANGED')
  48         elif s['tool_on'] or s['target_area'] or s['requested_area_m2']!=0:raise ValueError('CONNECTOR_HAS_WORK_CREDIT')
  49         previous=s['poses'][-1]
  50     return dict(passed=True,execution_segments=len(segments),reference_original_segments=len(old['segments']),original_reference_status='PARTIAL',original_work_target_union_m2=old_union.area,original_cluster_area_m2=old['work_area_m2'],original_work_union_unchanged=True)

  53 def load_checked(path=None):
  54     path=path or edition('R10')/'mowing_safe_plan.json'; plan=read(path)
  55     revision=plan.pop('revision'); computed=canonical(plan);plan['revision']=revision
  56     if computed!=revision:raise ValueError('SAFE_PLAN_REVISION_CHANGED')
  57     source=edition('R09')/'mission.json'
  58     if plan['source_sha256']!=sha(source):raise ValueError('REFERENCE_PLAN_CHANGED')
  59     for field,p in [('backend_sha256',APP/'build/r10_safe_route/r10_safe_route'),('backend_source_sha256',PKG/'r10_backend/safe_route.cpp'),('generator_sha256',PKG/'scripts/prepare_safe_plan_r10.py'),('envelope_source_sha256',PKG/'scripts/planning_envelope_r10.py')]:
  60         if sha(p)!=plan[field]:raise ValueError('SAFE_PLAN_GENERATOR_CHANGED '+field)
  61     validate(plan,read(source))
  62     return plan
