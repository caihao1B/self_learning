  31 def visibility(nodes,safe):
  32  corners=[]
  33  for poly in polygons(safe):
  34   for ring in [poly.exterior,*poly.interiors]:corners.extend(list(ring.coords)[:-1])
  35  xy=np.asarray([*nodes,*corners]);adj=[[] for _ in xy];allowed=safe.buffer(1e-8)
  36  for i in range(len(xy)):
  37   js=np.arange(i+1,len(xy))
  38   if not len(js):continue
  39   lines=shapely.linestrings(np.stack([np.tile(xy[i],(len(js),1)),xy[js]],axis=1));mask=shapely.covers(allowed,lines)
  40   for j in js[mask]:d=float(np.linalg.norm(xy[i]-xy[j]));adj[i].append((int(j),d));adj[j].append((i,d))
  41  costs=[];paths={}
  42  for i in range(len(nodes)):
  43   dist=[math.inf]*len(xy);dist[i]=0;prev={};q=[(0.,i)]
  44   while q:
  45    d,u=heapq.heappop(q)
  46    if d>dist[u]+1e-10:continue
  47    for v,w in adj[u]:
  48     if d+w<dist[v]-1e-10:dist[v]=d+w;prev[v]=u;heapq.heappush(q,(d+w,v))
  49   if not all(math.isfinite(x) for x in dist[:len(nodes)]):raise ValueError('DISCONNECTED_SERVICE_PATH')
  50   costs.append(dist[:len(nodes)])
  51   for j in range(len(nodes)):
  52    ids=[j]
  53    while ids[-1]!=i:ids.append(prev[ids[-1]])
  54    paths[i,j]=xy[list(reversed(ids))].tolist()
  55  return np.asarray(costs),paths
