   6 // Match progress scale to an explicitly validated, bounded precision motion.
   7 // The 10 s timeout, emergency priority and original work admission stay intact.
   8 inline double progressDelta(const nlohmann::json& cfg,const nlohmann::json& e) {
   9   const double coarse=cfg.at("progress_delta_m").get<double>();
  10   if (!cfg.contains("precision_progress_policy") ||
  11       e.value("motion_scope",std::string())!="VALIDATED_LOCAL_DOCKING" ||
  12       !e.contains("expected_translation_m") || !e["expected_translation_m"].is_number()) return coarse;
  13   const double distance=e["expected_translation_m"].get<double>();
  14   if (!std::isfinite(distance) || distance<=.008 || distance>.15) return coarse;
  15   return std::min(coarse,std::max(.005,.25*distance));
  16 }
