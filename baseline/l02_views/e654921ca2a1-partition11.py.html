  21 def kmeans(g,k,seed):
  22     xy=g['xy'];rng=np.random.default_rng(seed);ids=[int(rng.integers(len(xy)))];distance=((xy-xy[ids[0]])**2).sum(axis=1)
  23     for _ in range(1,k):
  24         j=int(rng.choice(len(xy),p=distance/distance.sum()));ids.append(j);distance=np.minimum(distance,((xy-xy[j])**2).sum(axis=1))
  25     centres=xy[ids].copy();labels=np.full(len(xy),-1,dtype=int)
  26     for _ in range(100):
  27         dist=((xy[:,None,:]-centres[None,:,:])**2).sum(axis=2);new=dist.argmin(axis=1)
  28         # Never silently reduce k after an empty Euclidean cluster.
  29         for label in range(k):
  30             if not np.any(new==label):
  31                 counts=np.bincount(new,minlength=k);eligible=np.where(counts[new]>1)[0]
  32                 j=int(eligible[np.argmax(dist[eligible,new[eligible]])]);new[j]=label
  33         if np.array_equal(labels,new):break
  34         labels=new;centres=np.array([xy[labels==i].mean(axis=0) for i in range(k)])
  35     return labels

  37 def repair(g,labels,k,c):
  38     labels=np.asarray(labels,dtype=int).copy();adj=g['adj'];weights=g['weights'];events=[]
  39     if len(labels)!=len(adj) or set(labels)!=set(range(k)):raise ValueError('EMPTY_OR_INVALID_LABELS')
  40     raw=[len(components(adj,np.flatnonzero(labels==i))) for i in range(k)]
  41     if max(raw)>1:
  42         # Retain each largest component. Grow only across a real shared edge.
  43         cores=[set(components(adj,np.flatnonzero(labels==i))[0]) for i in range(k)]
  44         assigned=np.full(len(labels),-1,dtype=int)
  45         for i,core in enumerate(cores):assigned[list(core)]=i
  46         loads=np.array([weights[list(core)].sum() for core in cores]);remaining=set(np.flatnonzero(assigned<0));target=weights.sum()/k
  47         while remaining:
  48             candidates=[]
  49             for u in sorted(remaining):
  50                 ns=sorted({int(assigned[v]) for v in adj[u] if assigned[v]>=0})
  51                 for dest in ns:candidates.append(((loads[dest]+weights[u])/target+.05*(dest!=labels[u]),u,dest))
  52             if not candidates:raise ValueError('REPAIR_DISCONNECTED_INPUT')
  53             _,u,dest=min(candidates);assigned[u]=dest;loads[dest]+=weights[u];remaining.remove(u)
  54         events.append(dict(kind='connected_region_growth',moved=int(np.count_nonzero(labels!=assigned))))
  55         labels=assigned
  56     loads=np.bincount(labels,weights=weights,minlength=k);mean=weights.sum()/k;total_edges=sum(e[2] for e in g['edges']);moves=0
  57     # Single-cell exchanges preserve both donor and receiver connectivity.
  58     for iteration in range(c['repair_balance_passes']):
  59         changed=0
  60         for u in sorted(range(len(adj)),key=lambda x:(-loads[labels[x]],x)):
  61             src=int(labels[u]);neighbors=adj[u]
  62             choices=[]
  63             for dst in sorted({int(labels[v]) for v in neighbors if labels[v]!=src}):
  64                 delta=3*((loads[src]-weights[u]-mean)**2+(loads[dst]+weights[u]-mean)**2-(loads[src]-mean)**2-(loads[dst]-mean)**2)/mean**2
  65                 cut_delta=sum(w*(int(dst!=labels[v])-int(src!=labels[v])) for v,w in neighbors.items())
  66                 score=delta+cut_delta/max(1,total_edges)
  67                 if score < -1e-10:choices.append((score,dst))
  68             if not choices:continue
  69             donor=set(np.flatnonzero(labels==src));donor.remove(u)
  70             if not donor or len(components(adj,donor))!=1:continue
  71             _,dst=min(choices);labels[u]=dst;loads[src]-=weights[u];loads[dst]+=weights[u];changed+=1
  72         moves+=changed
  73         if not changed:break
  74     events.append(dict(kind='connectivity_preserving_balance',moved=moves,passes=iteration+1))
  75     # Canonical IDs are sorted by computed centroid, not predetermined locations.
  76     order=sorted(range(k),key=lambda i:tuple(g['xy'][labels==i].mean(axis=0)))
  77     labels=np.array([order.index(int(i)) for i in labels],dtype=int)
  78     return labels,dict(raw_components=raw,events=events)
