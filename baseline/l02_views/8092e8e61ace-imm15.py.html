   1 """Causal interacting CV/CA/coordinated-turn filters.
   2 
   3 State is [x,y,vx,vy,ax,ay].
   4 Only causal position observations enter the update. Forecast modes are retained
   5 separately rather than collapsed into a constant-velocity average.
   6 """
   7 import math
   8 import numpy as np
   9 
  10 MODES = ('CV', 'CA', 'TURN_LEFT', 'TURN_RIGHT')
  11 
  12 
  13 def transition(mode, dt, turn_rate=.4):
  14     f = np.zeros((6, 6))
  15     f[:2, :2] = np.eye(2)
  16     if mode in ('CV', 'CA'):
  17         f[:2, 2:4] = np.eye(2)*dt
  18         f[2:4, 2:4] = np.eye(2)
  19         if mode == 'CA':
  20             f[:2, 4:] = np.eye(2)*(.5*dt*dt)
  21             f[2:4, 4:] = np.eye(2)*dt
  22             f[4:, 4:] = np.eye(2)
  23     else:
  24         omega = turn_rate if mode == 'TURN_LEFT' else -turn_rate
  25         s, c = math.sin(omega*dt), math.cos(omega*dt)
  26         f[:2, 2:4] = np.array([[s, c-1], [1-c, s]])/omega
  27         rotation = np.array([[c, -s], [s, c]])
  28         f[2:4, 2:4] = rotation
  29         f[4:, 2:4] = omega*np.array([[0, -1], [1, 0]])@rotation
  30     return f
  31

  33 def covariance(value, floor=1e-7):
  34     p = np.asarray(value, dtype=float)
  35     if p.shape[0] != p.shape[1] or not np.isfinite(p).all():
  36         raise ValueError('Invalid covariance')
  37     eigenvalues, eigenvectors = np.linalg.eigh((p+p.T)*.5)
  38     if eigenvalues.min() < -1e-5:
  39         raise ValueError('Non-PSD input covariance')
  40     return (eigenvectors*np.maximum(eigenvalues, floor))@eigenvectors.T
  41 
  42 
  43 def process_noise(dt, acceleration_sigma=.45):
  44     g = np.vstack([np.eye(2)*(.5*dt*dt), np.eye(2)*dt, np.eye(2)])
  45     return acceleration_sigma**2*dt*(g@g.T)+np.eye(6)*1e-7

  48 class IMM:
  49     def __init__(self, t, position, measurement_covariance, velocity=(0., 0.),
  50                  stay_probability=.94, acceleration_sigma=.45, turn_rate=.4):
  51         self.t = float(t)
  52         self.sigma, self.turn_rate = float(acceleration_sigma), float(turn_rate)
  53         if not math.isfinite(self.t) or not .5 < stay_probability < 1 or not 0 < self.turn_rate <= 2 or not math.isfinite(self.sigma) or self.sigma<=0:
  54             raise ValueError('Invalid IMM settings')
  55         x = np.array([*position, *velocity, 0., 0.], dtype=float)
  56         if x.shape != (6,) or not np.isfinite(x).all():
  57             raise ValueError('Invalid observation')
  58         p = np.diag([1., 1., .5, .5, 1., 1.])
  59         p[:2, :2] = covariance(measurement_covariance)
  60         self.means = np.tile(x, (4, 1))
  61         self.covs = np.tile(p, (4, 1, 1))
  62         self.probabilities = np.array([.45, .25, .15, .15])
  63         self.switch = np.full((4, 4), (1-stay_probability)/3)
  64         np.fill_diagonal(self.switch, stay_probability)
  65         self.updates = 1
  66 
  67     def update(self, t, position, measurement_covariance):
  68         dt = float(t)-self.t
  69         if not 0 < dt <= 1.2:
  70             raise ValueError('IMM observation must advance within the track freshness bound')
  71         z = np.asarray(position, dtype=float)
  72         r = covariance(measurement_covariance)
  73         if z.shape != (2,) or r.shape != (2, 2) or not np.isfinite(z).all():
  74             raise ValueError('Invalid measurement')
  75         predicted_probability = self.probabilities@self.switch
  76         mixing = self.probabilities[:, None]*self.switch/predicted_probability[None, :]
  77         means, covs, log_likelihoods = [], [], []
  78         h = np.eye(6)[:2]
  79         for j, mode in enumerate(MODES):
  80             x = np.sum(mixing[:, j, None]*self.means, axis=0)
  81             delta = self.means-x
  82             p = np.sum(mixing[:, j, None, None]*(self.covs+delta[:, :, None]*delta[:, None, :]), axis=0)
  83             f = transition(mode, dt, self.turn_rate)
  84             x = f@x
  85             p = f@p@f.T+process_noise(dt, self.sigma)
  86             innovation = z-h@x
  87             s = h@p@h.T+r
  88             gain = np.linalg.solve(s, h@p).T
  89             x = x+gain@innovation
  90             residual = np.eye(6)-gain@h
  91             p = residual@p@residual.T+gain@r@gain.T
  92             log_likelihoods.append(-.5*(innovation@np.linalg.solve(s, innovation)+np.linalg.slogdet(s)[1]+2*math.log(2*math.pi)))
  93             means.append(x)
  94             covs.append(covariance(p))
  95         log_probability = np.log(predicted_probability)+np.array(log_likelihoods)
  96         probability = np.exp(log_probability-log_probability.max())
  97         self.probabilities = probability/probability.sum()
  98         self.means, self.covs = np.array(means), np.array(covs)
  99         self.t, self.updates = float(t), self.updates+1

 101     def predict(self, horizon=3.2, dt=.4):
 102         if not 2 <= horizon <= 4 or not .1 <= dt <= 1 or abs(round(horizon/dt)*dt-horizon) > 1e-8:
 103             raise ValueError('Forecast needs 2–4 seconds with an integral number of steps')
 104         output = []
 105         for j, name in enumerate(MODES):
 106             x, p = self.means[j].copy(), self.covs[j].copy()
 107             f = transition(name, dt, self.turn_rate)
 108             means, covs = [], []
 109             for _ in range(round(horizon/dt)):
 110                 x = f@x
 111                 p = f@p@f.T+process_noise(dt, self.sigma)
 112                 means.append(x[:2].tolist())
 113                 covs.append(covariance(p[:2, :2]).tolist())
 114             output.append(dict(mode=name, probability=float(self.probabilities[j]),
 115                                mean=means, covariance=covs))
 116         return dict(t=self.t, frame='odom', dt=dt, horizon_s=horizon, modes=output,
 117                     algorithm='IMM_CV_CA_CT_LEFT_CT_RIGHT', updates=self.updates)
