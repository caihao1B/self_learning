   1 """Small differential-drive docking paths anchored once in measured local odometry.
   2 
   3 This is a motion proposal only. The caller must check the entire map-frame sweep,
   4 the live Nav2 costmap, measured stop and the original global work admission.
   5 """
   6 import math
   7 import statistics

  10 def wrap(a):
  11     return math.atan2(math.sin(a), math.cos(a))
  12 
  13 
  14 def ready_for_stopped_observation(samples,goal,now):
  15     """A local proximity proposal to cancel motion, never navigation/work credit.
  16 
  17     The original global stationary gate must still pass after the actual stop.
  18     Requiring a short fresh window avoids reacting to one noisy odometry sample.
  19     """
  20     if len(samples)<8 or samples[-1][0]-samples[0][0]<.25:return False
  21     if not -.03<=now-samples[-1][0]<.15:return False
  22     if any(not 0<b[0]-a[0]<=.15 for a,b in zip(samples,samples[1:])):return False
  23     return all(math.isfinite(t) and all(math.isfinite(v) for v in xy) and math.dist(xy[:2],goal[:2])<=.010 for t,xy in samples)
  24 
  25 
  26 def offset_observable(metrics):
  27     """Enough stationary evidence to propose a correction, never to grant work.
  28 
  29     A distance failure cannot disappear by collecting more of the same stable
  30     offset. Keep the original evidence and scatter limits; the work gate is
  31     still independently evaluated after the actual correction and stop.
  32     """
  33     keys=['samples','span_s','max_gap_s','position_scatter_p95_m','median_distance_m','latest_distance_m']
  34     if any(k not in metrics or not math.isfinite(metrics[k]) for k in keys):return False
  35     return (metrics['samples']>=20 and metrics['span_s']>=1.8 and
  36             metrics['max_gap_s']<=.15 and metrics['position_scatter_p95_m']<=.015 and
  37             metrics['median_distance_m']>.035 and metrics['latest_distance_m']>.035)
  38 
  39 
  40 def propose(history, target, global_pose, odom_pose, max_distance=.15):
  41     if len(history) < 20 or history[-1][0] - history[0][0] < 1.8:
  42         raise ValueError('INSUFFICIENT_STOPPED_OBSERVATIONS')
  43     if any(not all(math.isfinite(v) for v in p) for p in [target, global_pose, odom_pose]):
  44         raise ValueError('NONFINITE_POSE')
  45     median = [statistics.median(p[1][j] for p in history) for j in (0, 1)]
  46     if math.dist(global_pose[:2], median) > .03:
  47         raise ValueError('CURRENT_POSE_DISAGREES_WITH_STATIONARY_MEDIAN')
  48     delta = [target[j] - median[j] for j in (0, 1)]
  49     distance = math.hypot(*delta)
  50     if not .008 < distance <= max_distance:
  51         raise ValueError('CORRECTION_OUTSIDE_LOCAL_DOCKING_BOUND')
  52     bearing = math.atan2(delta[1], delta[0])
  53     turn = wrap(bearing - global_pose[2])
  54     rotation_steps = max(1, math.ceil(abs(turn) / .04))
  55     points = [[*global_pose[:2], global_pose[2] + turn * i / rotation_steps]
  56               for i in range(rotation_steps + 1)]
  57     steps = max(1, math.ceil(distance / .02))
  58     points += [[global_pose[0] + delta[0] * i / steps,
  59                 global_pose[1] + delta[1] * i / steps, bearing]
  60                for i in range(1, steps + 1)]
  61     angle = wrap(odom_pose[2] - global_pose[2])
  62     c, s = math.cos(angle), math.sin(angle)
  63     local = []
  64     for x, y, h in points:
  65         dx, dy = x - global_pose[0], y - global_pose[1]
  66         local.append([odom_pose[0] + c * dx - s * dy,
  67                       odom_pose[1] + s * dx + c * dy, wrap(h + angle)])
  68     return dict(map_path=points, odom_path=local, median=median,
  69                 correction_m=distance, frozen_map_to_odom_yaw=angle)
