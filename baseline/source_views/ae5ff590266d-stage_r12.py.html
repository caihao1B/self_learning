  10 def declared_scope_matches(receipt,spec,name):
  11  expected=CASES[name];fields={'seconds':'sim_seconds','distance_limit':'distance_limit_m'}
  12  if expected['task']=='mowing' and spec.get('segment_start',0)!=0:return False
  13  if receipt.get('case')!=name or receipt.get('task')!=expected['task'] or spec.get('task')!=expected['task']:return False
  14  if not all(spec.get(fields.get(k,k))==v for k,v in expected.items() if k!='task'):return False
  15  return expected['task']!='inspection' or spec.get('distance_limit_m')==expected.get('distance_limit',0.)
